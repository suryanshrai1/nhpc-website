// ERD version-1- Core
Table admins {
  id bigint [pk, increment]

  name varchar
  email varchar [unique]
  password_hash varchar

  created_at timestamp
  updated_at timestamp
}

Table media_folders {
  id bigint [pk, increment]

  name varchar
  slug varchar [unique]

  parent_folder_id bigint [null]

  created_at timestamp
  updated_at timestamp
}

Table media_files {
  id bigint [pk, increment]

  folder_id bigint

  file_name varchar
  original_name varchar

  mime_type varchar
  extension varchar

  file_size bigint

  storage_path varchar

  alt_text text

  uploaded_by bigint

  created_at timestamp
}

Ref: media_files.folder_id > media_folders.id

Ref: media_files.uploaded_by > admins.id

Ref: media_folders.parent_folder_id > media_folders.id
